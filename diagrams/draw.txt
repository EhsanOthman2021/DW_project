    %% ------------------------
    %% DATA SOURCES
    %% ------------------------
    subgraph DS[Data Sources]
        R1[Raw Invoice PDFs]
        R2[Raw Images (PNG/JPG)]
        R3[S3 Input Bucket<br/>my-fatura2-bucket]
    end

    %% ------------------------
    %% PIPELINE
    %% ------------------------
    subgraph PIPE[Data & ML Pipeline (DVC)]
        YAML[dvc.yaml]
        MAIN[main_pipline.py]
        CONF[config.json]

        P1[Step 1: Load & Preprocess<br/>clean text, extract metadata<br/>→ data/processed/*.json]
        P2[Step 2: Train ML Model<br/>XGBoost / features<br/>→ models/model_xgb.json + scaler.pkl]
        P3[Step 3: Evaluate<br/>→ reports/accuracy.html<br/>→ reports/roc_curve.html]
        P4[Step 4: Create Predictions<br/>→ test_with_predictions.csv]
    end

    %% ------------------------
    %% ARTIFACTS
    %% ------------------------
    subgraph ARTIFACTS[Artifacts Generated]
        A1[data/processed/*.json]
        A2[models/model_xgb.json]
        A3[models/scaler.pkl]
        A4[reports/*.html]
        A5[test_with_predictions.csv]
        CACHE[DVC Cache]
    end

    %% ------------------------
    %% API SERVICE
    %% ------------------------
    subgraph API[FastAPI Application<br/>src/api/service.py]
        LOADJ[Load invoice json files]
        LOADC[Load predictions CSV]
        MERGE[Merge metadata + true_label + predicted_label]
        END1[Endpoint: GET /]
        END2[Endpoint: GET /predict/{file_name}]
    end

    %% ------------------------
    %% DEPLOY
    %% ------------------------
    subgraph DEPLOY[Docker Deployment]
        DOCK[Dockerfile<br/>uvicorn src.api.service:app]
        COMP[docker-compose.yml<br/>AWS creds + S3 paths<br/>Volume mounts]
        S3D[(S3 Download<br/>invoice_model_reports/test_with_predictions.csv)]
        RUN[Running Container<br/>Port 8080]
    end
